COMMANDS FROM GUI

    MOVEMENT COMMAND
        - HALT
        - MOVE TO GOAL
        - LAND
        - TAKEOFF
        - MERGENCY LAND

        ros2 topic echo /cmd/control

    TARGET GOAL
        ros2 topic echo /cmd/goal
        ros2 topic pub --once /cmd/goal geometry_msgs/msg/PointStamped \
        "{header:{frame_id: map}, point:{x: 5.0, y: 1.0, z: 0.0}}"

    TARGET HEIGHT
        ros2 topic echo /cmd/height
        ros2 topic pub --once /cmd/height std_msgs/msg/Float32 "{data: 3.5}"

GOAL DETAILS

    GOAL POSITION
        # /goal/position  (geometry_msgs/PointStamped)
        ros2 topic pub --once /goal/position geometry_msgs/msg/PointStamped \
        "{header:{frame_id: map}, point:{x: 12.0, y: -4.0, z: 2.2}}"

    DISTANCE TO GOAL
        # /goal/distance  (std_msgs/Float32) — GUI shows “m”
        ros2 topic pub --once /goal/distance std_msgs/msg/Float32 "{data: 18.6}"

    TIME TO GOAL
        # /goal/time  (std_msgs/Float32) — total seconds (your GUI formats to h/m/s)
        ros2 topic pub --once /goal/time std_msgs/msg/Float32 "{data: 205.0}"
